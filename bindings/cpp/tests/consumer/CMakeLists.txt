cmake_minimum_required(VERSION 3.24)
project(ArboresceCppConsumer LANGUAGES CXX)
find_package(ArboresceCpp CONFIG REQUIRED)
set(ARBORESCE_LINKAGE shared CACHE STRING "C linkage")
option(ARBORESCE_NO_EXCEPTIONS "Disable consumer exceptions" OFF)
add_executable(consumer main.cpp)
if(ARBORESCE_LINKAGE STREQUAL "static")
    target_link_libraries(consumer PRIVATE Arboresce::cpp_static)
else()
    target_link_libraries(consumer PRIVATE Arboresce::cpp)
endif()
target_compile_options(consumer PRIVATE -Wall -Wextra -Werror)
if(ARBORESCE_NO_EXCEPTIONS)
    target_compile_options(consumer PRIVATE -fno-exceptions)
endif()
